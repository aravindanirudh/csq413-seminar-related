\section{Introduction}
%=====================================================

% ==================== SLIDE 3: BACKGROUND ====================
\begin{frame}{Background: Agent-Based Modeling (ABM)}
\begin{itemize}
    \item \textbf{Bottom-Up Paradigm:} Simulates autonomous entities (agents) interacting locally within a shared environment.
    \medskip
    \item \textbf{Emergence Principle:} Macro-level patterns emerge naturally from decentralized individual decisions.
    \medskip
    \item \textbf{Wide Applicability:} Used in traffic planning, epidemic spreading and financial market dynamics.
    \medskip
    \item \textbf{Core Objective:} Explains complex collective phenomena without requiring central control.
\end{itemize}
\end{frame}

% ==================== SLIDE 4: LIMITATIONS ====================
\begin{frame}{Limitations of Classical Behavioral Models}
\begin{itemize}
    \item \textbf{Strict Economic Rationality:} Assumes agents are perfectly rational utility maximizers (\textit{homo economicus}).
    \medskip
    \item \textbf{Absence of Emotional Nuance:} Cannot capture human stress, fear, empathy or shifting moods.
    \medskip
    \item \textbf{No Natural Language:} Communication is restricted to numbers and binary flags.
    \medskip
    \item \textbf{Rigid Decision Rules:} Fixed if-then heuristics fail when unprecedented situations arise.
    \medskip
    \item \textbf{Population Homogeneity:} Agents are grouped into simple types, ignoring individual diversity.
\end{itemize}
\end{frame}

% ==================== SLIDE 5: GENERATIVE SOCIAL SCIENCE VS GENAI ====================
\begin{frame}{The Concept of 'Generative' in Social Science}
\begin{itemize}
    \item \textbf{Epstein's Principle (1999):} \textit{"If you didn't grow it, you didn't explain it."}
    \medskip
    \item \textbf{Classical Meaning:} "Generative" meant growing macro-structures from simple local rules without central control.
    \medskip
    \item \textbf{Foundational Models:} Schelling's segregation model (1971) and Reynolds' flocking boids (1987).
    \medskip
    \item \textbf{The Technological Shift:} Classical models used hand-coded rules; modern GABMs use Large Language Models for cognitive reasoning.
\end{itemize}
\end{frame}

\section{Evolving Agents}
%=====================================================

% ==================== SLIDE 7: PRECURSORS: EAS AND RL ====================
\begin{frame}{Precursor Agent Architectures: EAs and RL}
\begin{itemize}
    \item \textbf{Evolutionary Algorithms (EAs):} Optimize agent rules via selection, crossover and mutation across generations.
    \medskip
    \item \textbf{EA Bottlenecks:} High computational cost, premature local convergence and heavy parameter tuning.
    \medskip
    \item \textbf{Reinforcement Learning (RL):} Optimizes numerical policies to maximize reward signals through trial and error.
    \medskip
    \item \textbf{RL Bottlenecks:} Sample inefficiency, multi-agent instability and total absence of language dialogue.
    \medskip
    \item \textbf{Takeaway:} Neither EAs nor RL provide human-like commonsense reasoning or open-ended communication.
\end{itemize}
\end{frame}

\section{Generative Agents}
%=====================================================

% ==================== SLIDE 10: DEFINING GAS ====================
\begin{frame}{Defining Generative Agents (GAs)}
\begin{itemize}
    \item \textbf{Formal Definition:} Autonomous computational entities powered by Large Language Models capable of human-like behavior and dialogue.
    \medskip
    \item \textbf{Mechanical Layer:} Rule-based physical substrate handling spatial coordinates, collisions and clock time.
    \medskip
    \item \textbf{Cognitive Layer:} LLM processes sensory perceptions, queries memory streams and decides actions.
    \medskip
    \item \textbf{Emergent Behaviors:} Decisions arise from pre-trained world knowledge rather than hand-coded if-then rules.
\end{itemize}
\end{frame}

% ==================== SLIDE 11: PARAMETERS VS STATUS ====================
\begin{frame}{Internal State: Static Parameters vs. Dynamic Status}
\begin{itemize}
    \item \textbf{Static Parameters (Identity Anchor):} Demographic data (age, gender, job), Big Five OCEAN personality traits and social network ties.
    \medskip
    \item \textbf{Dynamic Status (Runtime Representation):} Physical coordinates, current tasks, active goals and emotional affect.
    \medskip
    \item \textbf{Decoupled Architecture:} Separates internal state database storage from the LLM prompt engine.
    \medskip
    \item \textbf{Affective Tracking:} Records sentiment shifts, relationship changes and recent interaction history.
\end{itemize}
\end{frame}

% ==================== SLIDE 13: INFERENTIAL VS CONVERSATIONAL ====================
\begin{frame}{Inferential vs. Conversational Feedback Loops}
\begin{itemize}
    \item \textbf{Inferential Loop [Figure 2(a)]:} Agent receives an environmental event, checks internal status, and the LLM outputs a discrete physical action.
    \medskip
    \item \textbf{Environment Update:} Discrete actions directly alter physical variables in the simulation world.
    \medskip
    \item \textbf{Conversational Loop [Figure 2(b)]:} Agent receives dialogue from peers, updates internal beliefs and status, and outputs natural language text.
    \medskip
    \item \textbf{Social Communication:} Natural language responses directly influence the mental states of neighboring agents.
\end{itemize}
\end{frame}

% ==================== SLIDE 14: MEMORY STREAM & RETRIEVAL ====================
\begin{frame}{Core Cognitive Modules: Memory Stream \& Retrieval}
\begin{itemize}
    \item \textbf{Memory Stream:} Chronological database logging perceptions, dialogues and thoughts in natural language text.
\end{itemize}
\vspace{-0.15cm}
\begin{equation*}
\text{Score}(m, q) = \alpha \cdot \text{Recency}(m) + \beta \cdot \text{Importance}(m) + \gamma \cdot \text{Relevance}(m, q)
\end{equation*}
\vspace{-0.25cm}
\textbf{Symbol Definitions:}
\vspace{0.1cm}
\begin{itemize}
    \item $\text{Score}(m, q)$: Total retrieval priority for memory record $m$ under situational query $q$.
    \item $\alpha, \beta, \gamma$: Non-negative weights balancing recency, importance and relevance.
    \item $\text{Recency}(m) = e^{-\lambda \Delta t}$: Exponential decay with decay rate $\lambda$ and elapsed time $\Delta t$.
    \item $\text{Importance}(m)$: Integer score ($1$ to $10$) assigned by the LLM for event significance.
    \item $\text{Relevance}(m, q) = \frac{\mathbf{e}_m \cdot \mathbf{e}_q}{\|\mathbf{e}_m\| \|\mathbf{e}_q\|}$: Cosine similarity between embedding vectors $\mathbf{e}_m$ and $\mathbf{e}_q$.
\end{itemize}
\end{frame}

% ==================== SLIDE 15: REFLECTION & PLANNING ====================
\begin{frame}{Core Cognitive Modules: Reflection \& Dynamic Planning}
\begin{itemize}
    \item \textbf{Reflection Hierarchy:} Periodically synthesizes low-level memory records into higher-order abstract thoughts.
    \medskip
    \item \textbf{Belief Tree:} Forms an evolving directed tree from raw observations (leaves) to generalized beliefs (roots).
    \medskip
    \item \textbf{Recursive Planning:} Decomposes broad daily agendas into hourly schedules, then minute-level action sequences.
    \medskip
    \item \textbf{Dynamic Reactive Re-planning:} Pauses routine execution and updates plans when unexpected events occur.
\end{itemize}
\end{frame}

\section{Informal Theory of ABMs Validation}
%=====================================================

% ==================== SLIDE 18: VALIDATION DILEMMA ====================
\begin{frame}{The Fundamental Validation Dilemma}
\begin{itemize}
    \item \textbf{Purpose of Validation:} Ensures simulation models reflect real-world social processes rather than software bugs.
    \medskip
    \item \textbf{White-Box Traditional ABMs:} Model rules and transitions can be inspected and verified mathematically.
    \medskip
    \item \textbf{Black-Box Generative Agents:} Foundation models produce non-deterministic outputs from identical initial prompts.
    \medskip
    \item \textbf{Evidence Hierarchy:} Adapts evidence standards: controlled trials (highest), observational data (middle) and face validation (lowest).
\end{itemize}
\end{frame}

% ==================== SLIDE 19: FOUNDATIONAL & FACE VALIDATION ====================
\begin{frame}{Foundational and Face Validation in Traditional ABM}
\begin{itemize}
    \item \textbf{Data Analytics \& Docking:} Verifies distributions, correlations and compares outputs against independent models.
    \medskip
    \item \textbf{Empirical Testing \& Resampling:} Validates predictions on historical data and uses bootstrapping for confidence bounds.
    \medskip
    \item \textbf{Microface Validation:} Checks individual agent decisions and emotional logic for plausibility.
    \medskip
    \item \textbf{Macroface Validation:} Confirms collective group patterns (crowd flows, segregation) match real-world intuition.
    \medskip
    \item \textbf{Immersive Probing:} Domain experts enter virtual dashboards or VR to observe and interview agents.
\end{itemize}
\end{frame}

% ==================== SLIDE 21: PATHWAYS EXPLANATION ====================
\begin{frame}{Integrated Validation Pathways}
\begin{itemize}
    \item \textbf{Empirical Pathway (Quantitative Matching):} Conditions initial demographics with census data and tests aggregate trajectories against historical data.
    \medskip
    \item \textbf{Model Alignment:} Applies data analytics and docking across runs to ensure numerical reliability.
    \medskip
    \item \textbf{Macroface Pathway (Collective Patterns):} Validates emergent group dynamics such as spatial clustering and norm spread without central control.
    \medskip
    \item \textbf{Microface Pathway (Individual Realism):} Probes individual agents through structured interviews to check reasoning, memories and consistency.
\end{itemize}
\end{frame}

% ==================== SLIDE 22: AUTOMATED VALIDATION ====================
\begin{frame}{Automated Validation: LLM-as-a-Judge \& Algorithmic Fidelity}
\begin{itemize}
    \item \textbf{LLM-as-a-Judge:} Uses independent LLMs to interview agents and grade multi-agent dialogues at large scale.
    \medskip
    \item \textbf{Evaluation Strengths:} High accuracy on objective goals, factual consistency and quantitative criteria.
    \medskip
    \item \textbf{Evaluation Limits:} Struggles with subtle social nuances, passive aggression, sarcasm and deception.
    \medskip
    \item \textbf{Algorithmic Fidelity:} Measures how accurately simulated personas reflect genuine human demographic attitudes.
    \medskip
    \item \textbf{Similarity Principle:} Rigorous testing on sample scenarios builds confidence across related unobserved spaces.
\end{itemize}
\end{frame}

\section{GABMs: Emerging Challenges}
%=====================================================

% ==================== SLIDE 25: HALLUCINATIONS ====================
\begin{frame}{Challenge 1: Large Language Model Hallucinations}
\begin{itemize}
    \item \textbf{Factuality Hallucinations:} Generating assertions that contradict verified physical environment facts.
    \medskip
    \item \textbf{Faithfulness Hallucinations:} Diverging from assigned persona values, moral codes or prior conversation history.
    \medskip
    \item \textbf{Data Stage Mitigation:} Grounding prompts with Retrieval-Augmented Generation (RAG) using live environment facts.
    \medskip
    \item \textbf{Training Stage Mitigation:} Applying Reinforcement Learning from Human Feedback (RLHF) and persona fine-tuning.
    \medskip
    \item \textbf{Inference Stage Mitigation:} Enforcing constrained grammar decoding (e.g., JSON schema) and multi-agent peer verification.
\end{itemize}
\end{frame}

% ==================== SLIDE 26: COMPUTATIONAL BOTTLENECKS ====================
\begin{frame}{Challenge 2: Computational Bottlenecks and Scaling}
\begin{itemize}
    \item \textbf{Computational Bottlenecks:} Quadratic attention latency and pairwise agent communication scaling:
\end{itemize}
\vspace{-0.15cm}
\begin{equation*}
\text{Complexity}_{\text{Attention}} = \mathcal{O}(n^2 \cdot d), \quad \text{Complexity}_{\text{Communication}} = \mathcal{O}(N^2)
\end{equation*}
\vspace{-0.25cm}
\textbf{Symbol Definitions:}
\vspace{0.1cm}
\begin{itemize}
    \item $n$: Sequence length (number of input context tokens in memory prompt).
    \item $d$: Embedding representation dimension of the Transformer model.
    \item $N$: Total number of interacting generative agents in the simulation.
    \item $M$: Number of memory records searched during vector retrieval ($\mathcal{O}(\log M)$ to $\mathcal{O}(M)$).
    \item \textbf{The Scalability Gap:} Traditional ABMs run $10^6$ agents; pure GABMs remain limited to dozens or hundreds.
\end{itemize}
\end{frame}

\section{Discussion}
%=====================================================

% ==================== SLIDE 27: CITIES AS TESTBEDS ====================
\begin{frame}{Discussion: Cities as a Natural Testbed for GABMs}
\begin{itemize}
    \item \textbf{Alignment with Real-World Text:} Urban laws, historical archives and census data closely match LLM training text.
    \medskip
    \item \textbf{Beyond Physical Metrics:} Traditional urban models measure only functional data (traffic delay, throughput, capacity).
    \medskip
    \item \textbf{Qualitative Experience Mapping (QEM):} Models subjective human feelings: safety perceptions in unlit streets, comfort and transit stress.
    \medskip
    \item \textbf{Human-Centered Planning:} Informs urban policy to support community emotional well-being and pedestrian safety.
\end{itemize}
\end{frame}

% ==================== SLIDE 28: HYBRID ARCHITECTURE ====================
\begin{frame}{The Proposed Solution: Combined Hybrid ABM-GABM}
\begin{itemize}
    \item \textbf{Complementary Synergy:} Unifies the high computational speed of rule-based ABMs with the cognitive depth of GABMs.
    \medskip
    \item \textbf{Low-Level Mechanistic Layer (Traditional ABM):} Handles deterministic, high-frequency physics: collision avoidance and kinematics ($\mathcal{O}(1)$ or $\mathcal{O}(N)$).
    \medskip
    \item \textbf{High-Level Cognitive Layer (Generative Agents):} Invoked only during critical decision points: moral choices, negotiations and social cooperation.
    \medskip
    \item \textbf{Key Benefit:} Resolves computational bottlenecks while preserving rich human-like behavior.
\end{itemize}
\end{frame}

% ==================== SLIDE 29: CASE STUDY & VALIDATION ====================
\begin{frame}{Case Study: Building Evacuation \& Hybrid Validation}
\begin{itemize}
    \item \textbf{Emergency Building Fire Evacuation:} Illustrates the division of labor in an emergency crisis.
    \medskip
    \item \textbf{Low-Level Rule Engine:} Moves agents through corridors, stairwells and exit doors using fast physics heuristics.
    \medskip
    \item \textbf{High-Level LLM Engine:} Decides whether an agent stops to help an injured peer or flees to save itself.
    \medskip
    \item \textbf{Macro-Validation:} Sensor data confirms crowd evacuation clearance times match building drill records.
    \medskip
    \item \textbf{Micro-Validation:} Post-simulation interviews verify individual survival reasoning and moral choices.
\end{itemize}
\end{frame}
